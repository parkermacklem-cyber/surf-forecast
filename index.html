/* TRADESCAN V4 Quantitative Trading Research Engine Features: - 20 stock watchlist - EMA 9 / 20 / 50 - RSI 14 - MACD - ATR 14 - Bollinger Bands - Relative volume - Momentum analysis - Trend detection - 0-100 weighted score - Signal explanations - Paper trade simulation - Risk / reward calculations */ const API = "https://trade-scanner-api.parkermacklem.workers.dev"; const WATCHLIST = [ "AAPL", "MSFT", "NVDA", "AMZN", "GOOGL", "META", "AVGO", "TSLA", "JPM", "V", "MA", "WMT", "COST", "NFLX", "AMD", "ORCL", "CRM", "BAC", "XOM", "LLY" ]; const CONFIG = { interval: "5min", outputsize: 500, startingCash: 10000, riskPercent: 1, maxPositionPercent: 20, commission: 0, slippagePercent: 0.1, weights: { trend: 0.25, momentum: 0.20, macd: 0.15, rsi: 0.15, volume: 0.10, volatility: 0.10, structure: 0.05 } }; // ---------- UTILITIES ---------- function average(values) { if (!values.length) return 0; return values.reduce((a, b) => a + b, 0) / values.length; } function clamp(value, min, max) { return Math.max(min, Math.min(max, value)); } function last(values) { return values[values.length - 1]; } function round(value, decimals = 2) { return Number(value.toFixed(decimals)); } function percentChange(current, previous) { if (!previous) return 0; return ((current - previous) / previous) * 100; } function standardDeviation(values) { const mean = average(values); const variance = average( values.map(v => Math.pow(v - mean, 2)) ); return Math.sqrt(variance); } // ---------- TECHNICAL INDICATORS ---------- function calculateEMA(values, period) { if (!values.length) return []; const multiplier = 2 / (period + 1); let previous = values[0]; return values.map((price, index) => { if (index === 0) return previous; previous = price * multiplier + previous * (1 - multiplier); return previous; }); } function calculateSMA(values, period) { return values.map((_, index) => { if (index + 1 < period) return null; return average( values.slice(index + 1 - period, index + 1) ); }); } function calculateRSI(values, period = 14) { if (values.length < period + 1) return 50; let gains = 0; let losses = 0; for (let i = values.length - period; i < values.length; i++) { const change = values[i] - values[i - 1]; if (change > 0) gains += change; else losses -= change; } if (losses === 0) return gains === 0 ? 50 : 100; const relativeStrength = gains / losses; return 100 - 100 / (1 + relativeStrength); } function calculateMACD(values) { const fast = calculateEMA(values, 12); const slow = calculateEMA(values, 26); const macdLine = fast.map( (value, index) => value - slow[index] ); const signalLine = calculateEMA(macdLine, 9); return { macd: last(macdLine), signal: last(signalLine), histogram: last(macdLine) - last(signalLine) }; } function calculateATR(candles, period = 14) { const trueRanges = []; for (let i = 1; i < candles.length; i++) { const current = candles[i]; const previous = candles[i - 1]; const range = Math.max( current.high - current.low, Math.abs(current.high - previous.close), Math.abs(current.low - previous.close) ); trueRanges.push(range); } return average(trueRanges.slice(-period)); } function calculateBollingerBands(values, period = 20) { const recent = values.slice(-period); const middle = average(recent); const deviation = standardDeviation(recent); return { upper: middle + deviation * 2, middle, lower: middle - deviation * 2 }; } function calculateMomentum(values, bars) { if (values.length <= bars) return 0; return percentChange( last(values), values[values.length - 1 - bars] ); } function calculateRelativeVolume(candles) { const volumes = candles.map(c => c.volume); const recent = average(volumes.slice(-5)); const baseline = average(volumes.slice(-50, -5)); return baseline > 0 ? recent / baseline : 1; } // ---------- MARKET DATA ---------- async function fetchMarketData(symbol) { const url = API + "/timeseries?symbol=" + encodeURIComponent(symbol) + "&interval=" + CONFIG.interval + "&outputsize=" + CONFIG.outputsize; const response = await fetch(url, { cache: "no-store" }); if (!response.ok) { throw new Error( symbol + ": HTTP " + response.status ); } const data = await response.json(); if (data.error) { throw new Error(data.details || data.error); } if (!Array.isArray(data.candles)) { throw new Error("Missing candle data"); } const candles = data.candles.map(c => ({ open: Number(c.open), high: Number(c.high), low: Number(c.low), close: Number(c.close), volume: Number(c.volume) || 0, datetime: c.datetime })); if (candles.length < 50) { throw new Error("Insufficient market history"); } return candles; } // ---------- FEATURE EXTRACTION ---------- function extractFeatures(symbol, candles) { const closes = candles.map(c => c.close); const price = last(closes); const ema9 = last(calculateEMA(closes, 9)); const ema20 = last(calculateEMA(closes, 20)); const ema50 = last(calculateEMA(closes, 50)); const rsi = calculateRSI(closes); const macd = calculateMACD(closes); const atr = calculateATR(candles); const bands = calculateBollingerBands(closes); const relativeVolume = calculateRelativeVolume(candles); const momentum15 = calculateMomentum(closes, 3); const momentum30 = calculateMomentum(closes, 6); const momentum60 = calculateMomentum(closes, 12); const recent = candles.slice(-20); const resistance = Math.max( ...recent.map(c => c.high) ); const support = Math.min( ...recent.map(c => c.low) ); const rangePosition = resistance === support ? 50 : ((price - support) / (resistance - support)) * 100; const atrPercent = (atr / price) * 100; return { symbol, candles, price, ema9, ema20, ema50, rsi, macd, atr, atrPercent, bands, relativeVolume, momentum15, momentum30, momentum60, support, resistance, rangePosition }; } // ---------- MARKET REGIME ---------- function detectMarketRegime(f) { if ( f.price > f.ema9 && f.ema9 > f.ema20 && f.ema20 > f.ema50 ) { return "BULLISH TREND"; } if ( f.price < f.ema9 && f.ema9 < f.ema20 && f.ema20 < f.ema50 ) { return "BEARISH TREND"; } if (f.atrPercent > 3) { return "HIGH VOLATILITY"; } return "MIXED / SIDEWAYS"; } // ---------- QUANT SCORING ENGINE ---------- function calculateScores(f) { const trend = (f.price > f.ema9 ? 30 : 0) + (f.ema9 > f.ema20 ? 35 : 0) + (f.ema20 > f.ema50 ? 35 : 0); const momentum = clamp( 50 + f.momentum15 * 8 + f.momentum30 * 4 + f.momentum60 * 2, 0, 100 ); const macd = clamp( 50 + (f.macd.histogram / f.price) * 3000, 0, 100 ); const rsi = clamp( 100 - Math.abs(f.rsi - 57) * 2.2, 0, 100 ); const volume = clamp( 45 + (f.relativeVolume - 1) * 40, 0, 100 ); const volatility = clamp( 90 - Math.max(0, f.atrPercent - 1.2) * 25, 0, 100 ); const structure = clamp( f.rangePosition, 0, 100 ); const parts = { trend, momentum, macd, rsi, volume, volatility, structure }; let total = 0; for (const key in CONFIG.weights) { total += parts[key] * CONFIG.weights[key]; } return { total: Math.round(clamp(total, 0, 100)), parts }; } // ---------- SIGNAL EXPLANATION ---------- function generateExplanation(f, score) { const reasons = []; if (f.ema9 > f.ema20) { reasons.push("Short-term EMA trend is positive"); } else { reasons.push("Short-term EMA trend is weak"); } if (f.macd.histogram > 0) { reasons.push("MACD momentum is positive"); } else { reasons.push("MACD momentum is negative"); } if (f.rsi > 70) { reasons.push("RSI indicates overbought conditions"); } else if (f.rsi < 30) { reasons.push("RSI indicates oversold conditions"); } else { reasons.push("RSI is within its middle range"); } if (f.relativeVolume > 1.5) { reasons.push("Trading volume is elevated"); } if (f.atrPercent > 3) { reasons.push("Price volatility is elevated"); } return { classification: score >= 80 ? "STRONG TECHNICAL SCORE" : score >= 65 ? "POSITIVE TECHNICAL SCORE" : score >= 50 ? "MIXED TECHNICAL SCORE" : "WEAK TECHNICAL SCORE", reasons }; } // ---------- PAPER TRADING RISK ENGINE ---------- function calculatePaperTrade(f) { const entry = f.price; const invalidation = Math.max( 0, Math.min( entry - f.atr * 1.35, f.support - f.atr * 0.25 ) ); const riskPerShare = entry - invalidation; const accountRisk = CONFIG.startingCash * (CONFIG.riskPercent / 100); const maxPositionValue = CONFIG.startingCash * (CONFIG.maxPositionPercent / 100); const sharesByRisk = riskPerShare > 0 ? Math.floor(accountRisk / riskPerShare) : 0; const sharesByCapital = Math.floor(maxPositionValue / entry); const shares = Math.max( 0, Math.min(sharesByRisk, sharesByCapital) ); const targets = { oneR: entry + riskPerShare, twoR: entry + riskPerShare * 2, threeR: entry + riskPerShare * 3 }; const positionValue = shares * entry; return { entry: round(entry), invalidation: round(invalidation), riskPerShare: round(riskPerShare), shares, positionValue: round(positionValue), maximumPlannedRisk: round( shares * riskPerShare ), targets: { oneR: round(targets.oneR), twoR: round(targets.twoR), threeR: round(targets.threeR) } }; } // ---------- FULL ANALYSIS ---------- function analyzeStock(symbol, candles) { const features = extractFeatures(symbol, candles); const regime = detectMarketRegime(features); const scoring = calculateScores(features); const explanation = generateExplanation( features, scoring.total ); const paperTrade = calculatePaperTrade(features); return { symbol, price: round(features.price), score: scoring.total, regime, classification: explanation.classification, reasons: explanation.reasons, indicators: { ema9: round(features.ema9), ema20: round(features.ema20), ema50: round(features.ema50), rsi: round(features.rsi), macd: round(features.macd.macd, 4), macdSignal: round(features.macd.signal, 4), macdHistogram: round( features.macd.histogram, 4 ), atr: round(features.atr), atrPercent: round(features.atrPercent), relativeVolume: round(features.relativeVolume), momentum15: round(features.momentum15), momentum30: round(features.momentum30), momentum60: round(features.momentum60), support: round(features.support), resistance: round(features.resistance), bollingerUpper: round(features.bands.upper), bollingerLower: round(features.bands.lower) }, scoreBreakdown: scoring.parts, paperTrade }; } // ---------- SCAN WATCHLIST ---------- async function scanWatchlist() { const results = []; const errors = []; console.log("Starting TradeScan V4..."); console.log("Scanning", WATCHLIST.length, "stocks"); for (const symbol of WATCHLIST) { try { const candles = await fetchMarketData(symbol); const analysis = analyzeStock( symbol, candles ); results.push(analysis); console.log( symbol, "Price:", analysis.price, "Score:", analysis.score, "Regime:", analysis.regime ); } catch (error) { errors.push({ symbol, error: error.message }); console.error(symbol, error.message); } } results.sort((a, b) => b.score - a.score); console.table( results.map(stock => ({ Symbol: stock.symbol, Price: stock.price, Score: stock.score, Regime: stock.regime })) ); console.log("Scan completed"); console.log("Successful:", results.length); console.log("Errors:", errors.length); return { results, errors }; } // ---------- PAPER TRADE SIMULATION ---------- function simulatePaperTrade(analysis, exitPrice) { const trade = analysis.paperTrade; const shares = trade.shares; const entryCost = trade.entry * shares * (1 + CONFIG.slippagePercent / 100); const exitValue = exitPrice * shares * (1 - CONFIG.slippagePercent / 100); const profit = exitValue - entryCost - CONFIG.commission * 2; const returnPercent = entryCost > 0 ? profit / entryCost * 100 : 0; return { symbol: analysis.symbol, shares, entry: trade.entry, exit: exitPrice, profit: round(profit), returnPercent: round(returnPercent), outcome: profit > 0 ? "PROFIT" : profit < 0 ? "LOSS" : "BREAKEVEN" }; } // ---------- RUN SCANNER ---------- scanWatchlist().then(scan => { window.tradeScanResults = scan.results; window.tradeScanErrors = scan.errors; console.log( "Results available at window.tradeScanResults" ); });
